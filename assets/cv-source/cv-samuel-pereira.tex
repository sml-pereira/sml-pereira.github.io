% Section heading — matched to the website's h2
%   colour   accent, not secondary
%   tracking 0.12em (LetterSpace=12), twice the previous setting
%   rule     accenthl fading to transparent, and only as wide as the heading
%            text — on the site the underline is a block inside an inline-block
%            h2, so it measures the words rather than a fixed fraction.
%   The fleuron is gone; the website has no such ornament.
\newsavebox{\cvsectionbox}
\newcommand{\cvsection}[1]{%
    \needspace{4\baselineskip}%  keeps a heading from being orphaned at a page foot
    \vspace{\secsep}%
    \sbox{\cvsectionbox}{\sffamily\scriptsize\addfontfeature{LetterSpace=12}\color{accent}\MakeUppercase{#1}}%
    \usebox{\cvsectionbox}%
    \par\vspace{3pt}%
    \begin{tikzpicture}
        \fill[accenthl, path fading=east] (0,0) rectangle (\wd\cvsectionbox, 0.4pt);
    \end{tikzpicture}%
    \par\vspace{\spacesmall}%
}

% Subsection — the website's "kind" label: small uppercase Inter in tertiary
% with wide tracking, not the italic serif this used to be.
% \removelastskip discards whatever vertical glue precedes the label — an
% entry's \entrysep or a section heading's \spacesmall — so \subsecsep sets
% the gap absolutely rather than adding to it. LaTeX has no margin collapsing,
% so without this the same \subsecsep produced one gap under a heading and a
% much larger one after entries. Every subsection label now opens at exactly
% \spacesmall, whatever precedes it, and closes at \spacesmall too.
\newcommand{\cvsubsection}[1]{%
    \par\removelastskip
    \vspace{\subsecsep}%
    {\sffamily\fontsize{7}{9}\selectfont\addfontfeature{LetterSpace=10}%
     \cvdot\color{tertiary}\MakeUppercase{#1}}%
    \par\vspace{\spacesmall}%
}

% === RESEARCH PROFILE ===
I'm a researcher in music theory and computational musicology developing the Fourier Qualia Space, a novel framework applying discrete Fourier transforms to model harmonic-quality motion and ambiguity across four centuries of Western art music, with a special focus on post-tonal repertoires. My work combines quantitative analysis with music-theoretical enquiry, bridging computational methods and humanistic interpretation to understand how harmonic qualities convey hierarchical and syntactic organisation.
